		b_diff_slice = (1L*scale(dac_mid,acq_z[b_calc_cnt],1000)*grad_var[0])/32767L;
		templ3 = b_diff_read*b_diff_read+b_diff_phase*b_diff_phase+b_diff_slice*b_diff_slice;
		b_trial = (((templ3+5000L)/10000L)*b_kfac + 5000L)/10000L; // [MERGE-2TE] F3 kernel
		if ((max_diff_grad_pc==1)&&(report_on==1)&&(b_target>0))
			printf("b[%d]: want %ld, got %ld, DAC %d\n", b_calc_cnt, b_target, b_trial, dac_lo);
		if (b_trial > b_target)
			templ3 = b_trial-b_target;
		else
			templ3 = b_target-b_trial;
		if ((max_diff_grad_pc==1)&&(b_target>0)&&(templ3*100L > b_target*5L))
			printf("b[%d] differs >5%%; record achieved b\n", b_calc_cnt);
		b_calc_cnt = b_calc_cnt+1;
		if (b_calc_cnt >= no_diff_acq)
			goto b_value_mode_done;
		goto b_value_entry_loop;
b_value_mode_done:
		// [V17-REVIEW] scale() returns a 16-bit int. Check the LONG result
		// before runtime scale() can narrow it (e.g. 20000 DAC at 200%).
		b_calc_cnt=0;
diffusion_scale_check:
		templ1=(IntToLong(acq_grad[b_calc_cnt])*IntToLong(diff_scale_saved))/100L;
		if ((acq_grad[b_calc_cnt]<0)||((diff_on==1)&&(templ1>30000L)))
		{ v19_error_code=51; goto end; }
		if (templ1>32767L)
		{ v19_error_code=52; goto end; }
		b_calc_cnt=b_calc_cnt+1;
		if (b_calc_cnt<no_diff_acq) goto diffusion_scale_check;
		// [B-VALUE MODE] end

	
/*   ********************** PHASE ENCODE MATRIX CALCULATION ***************************** */



	if ((no_views<1)||(no_views>MAXPESIZE)||(views_per_seg<1)||(views_per_seg>no_views)) 
	{
		v19_error_code=53;
		goto end;
	}


	//  Check that the base echo time is a multiple of the effective echo time

	/*
	if (te%esp != 0 ) 	{
		printf("The base echo time must be a multiple of the eff echo time \n");
		goto end;
	}
	*/
	if(diff_on == 0)
	{
		if(esp == 0)
		{
			v19_error_code=54;
			goto end;
		}
		te_eff = scale(te, 1, esp);
	}	
	else
	{
		// [CRUSH-PE] DWI timing always uses te as FIRST echo, independent of ordering.
		te_eff = 1;
		if (PE_order != 1)
		{
			if ((PE_order != 5)&&(PE_order != 6)&&(PE_order != 7))
			{
				v19_error_code=55;
				goto end;
			}
		}
		/* [MERGE-2TE] M5 (port): the DE (-90 flip-back) timing is the legacy
		   esp-based chain (pre_90_delay0L, te_balance_bl) and is not valid on the
		   two-TE skeleton - block the combination instead of playing a silently
		   mistimed flip-back. */
		if (de_on == 1)
		{
			v19_error_code=56;
			goto end;
		}
	}
	// [CRUSH-PE] New orders are full-Fourier multi-echo DWI only.
	if ((PE_order==6)||(PE_order==7))
	{
		if ((diff_on!=1)||(views_per_seg<=1)||(echoes_to_discard!=0))
		{
			v19_error_code=57;
			goto end;
		}
		if ((PF_echoes!=0)&&(PF_echoes!=views_per_seg))
		{
			v19_error_code=58;
			goto end;
		}
	}
	if ((PE_order<0)||(PE_order>7)||(PE_order==3)||(PE_order==4))
	{
		v19_error_code=59;
		goto end;
	}
	if (PE_order != 2){
		no_views_eff = no_views;
		if(nav_on==1) no_views_eff = no_views - views_per_seg;
		
		if (no_views_eff<1)
		{
			v19_error_code=60;
			goto end;
		}
		if ((no_views_eff%2 != 0 )&&(PE_order != 5)) {
			v19_error_code=61;
			goto end;
		}
		
		if (PF_echoes == 0) PF_echoes = views_per_seg;
		
		if (PF_echoes < views_per_seg )  {
			v19_error_code=62;
			goto end;
		} 

		if (PF_echoes >= 2*views_per_seg )  {
			v19_error_code=63;
			goto end;
		} 
		views_per_echo = no_views_eff/(2*views_per_seg);
		view_shift = (PF_echoes - views_per_seg) * views_per_echo;		
	}
	else {
		// For single shot, only acquire part of the k-space
		// the number of lines in the full k-space is no_views_eff
		// But we still only acquire no_views echoes
		// no_views_eff - no_views is the number of missing lines
		no_views_eff = (no_views - te_eff + 1)*2;

		
		if (no_views<=te_eff)  {
			v19_error_code=64;
			goto end;
		}
		view_shift = 0;		
	}	

	
#ifdef SKIP_FIRST_ECHOES
	te_eff = te_eff-echoes_to_discard;
	if (report_on) printf("First %d echoes discarded: ", echoes_to_discard);
	if (te_eff<0) 
	{
		v19_error_code=65;
		goto end;
	}
	else
	{
		if (report_on)  printf("te_eff=%d\n", te_eff);
	}
#endif

	if (te_eff > views_per_seg ) 	{
		v19_error_code=66;
		goto end;
	}

	cnt = 0;

	if (nav_on) {
		navigator_loop:
		if (cnt<views_per_seg) {
			//gp_order[cnt] = 0;
			put_gp_order(cnt,0);
			cnt = cnt+1;
			goto navigator_loop;
		}
	}	



	if (views_per_seg == 1) // single echo
	{
		if(PE_order != 5) 
		{
			v19_error_code=67;
			goto end;
		}
	}
	
	if (PE_order == 5) // single echo
	{
		if(views_per_seg != 1) 
		{
			v19_error_code=68;
			goto end;
		}
	}
	
	
    if ((PE_order<2)||(PE_order==6))
	{
		if (no_views_eff%(views_per_seg*2) != 0 ) 	
		{
			v19_error_code=69;
			goto end;
		}
	}	

	if ((PE_order==7)&&(no_views_eff%views_per_seg!=0))
	{
		v19_error_code=70;
		goto end;
	}
	if ((PE_order==6)||(PE_order==7))
		goto PE_order_6_7;
    if (PE_order == 0)
		goto PE_order_0;
	if (PE_order == 1) 
		goto PE_order_1;
	if (PE_order == 2) 
		goto PE_order_2;
	
	if (PE_order == 5) 
		goto PE_order_5;
	
	
PE_order_0: 

    // Use pfgen.exe to create the reconstruction table

	// Check that phase encode and echo parameters are compatible
	if ((PE_order == 0)&&(views_per_seg<4))
	{
		v19_error_code=71;
		goto end;
	}

	// [CRUSH-SCHED] PE0 uses one-based doubled scratch tables: index 512
	// is outside 0..511, even when its aliased location word is unused.
	if ((2L*IntToLong(views_per_seg)>=MAXCENTRE)||(2L*IntToLong(views_per_echo)>=MAXLOCATION))
	{
		v19_error_code=72;
		goto end;
	}
	if (2L*IntToLong(views_per_seg)*IntToLong(views_per_echo)>MAXPESIZE)
	{
		v19_error_code=73;
		goto end;
	}

	//  Set up the echo_centre array
	array_count = 0;
echo_centre_loop:   
    array_count = array_count + 1;

	gp_mul = (array_count-te_eff)*views_per_echo-1;
	if(gp_mul < (-no_views_eff/2) - view_shift)
		gp_mul = gp_mul + no_views_eff;
	if(gp_mul >= (no_views_eff/2) - view_shift)
		gp_mul = gp_mul - no_views_eff;
	
	//echo_centre[array_count] = gp_mul;
	put_echo_centre(array_count, gp_mul);

	gp_mul = (1-te_eff-array_count)*views_per_echo-1;
	if(gp_mul < (-no_views_eff/2) - view_shift) 
		gp_mul = gp_mul + no_views_eff;
	if(gp_mul >= (no_views_eff/2) - view_shift) 
		gp_mul = gp_mul - no_views_eff;
	
	//echo_centre[array_count + views_per_seg] = gp_mul;
	put_echo_centre(array_count + views_per_seg, gp_mul);

    if(array_count < views_per_seg)
		goto echo_centre_loop;

	// Set up the echo_location array
	array_count = 0;

echo_location_loop:   
	array_count = array_count + 1;
	
	//echo_location[array_count] = (views_per_echo/2) - array_count + 1;
	//echo_location[array_count+views_per_echo] = echo_location[array_count];	
	temp = (views_per_echo/2) - array_count + 1;	
	put_echo_location(array_count, temp);
	put_echo_location(array_count+views_per_echo, temp);

	if(array_count < views_per_echo) 
		goto echo_location_loop;	 
	 
	/*  Set up the gp_order array */
	
	array_count = cnt;
    gp_loc = 0;
    gp_store = 0;	
	
gp_order_loop1:
    gp_loc = gp_loc + 1;
    if (gp_loc > views_per_echo) 
		gp_store = views_per_seg;
    gp_cnt = 0;
gp_order_loop2:
    gp_cnt = gp_cnt + 1;
    //gp_mul = echo_centre[gp_cnt + gp_store] + echo_location[gp_loc];
	gp_mul = get_echo_centre(gp_cnt + gp_store)+get_echo_location(gp_loc);
	
	
    if(gp_mul < (-no_views_eff/2)-view_shift) 
		gp_mul = gp_mul + no_views_eff;
    if(gp_mul >= (no_views_eff/2)-view_shift) 
		gp_mul = gp_mul - no_views_eff;
    //gp_order[array_count] = gp_mul;
	put_gp_order(array_count,gp_mul);
    array_count = array_count + 1;  
    if (gp_cnt < views_per_seg) 
		goto gp_order_loop2;
    if (gp_loc < 2*views_per_echo) 
		goto gp_order_loop1;
	goto setup_image_params;
	
PE_order_1:

	// Use egen.exe to create the reconstruction table
	
	shot = 1;
	// Number of shots is (views_per_echo*2)
	no_views_eff_1 = no_views_eff;
	gp_loc = no_views_eff_1/2-views_per_echo-1;
	shots_loop_3:	
	
		echo = 1;
		gp_loc = gp_loc+1;
		echoes_loop_3:
			if (shot>views_per_echo) 
				gp_mul = gp_loc+views_per_echo*(echo-te_eff);
			else 
				gp_mul = gp_loc-views_per_echo*(echo-te_eff);
			
			//gp_order[cnt] = gp_mul - no_views_eff_1/2;
			put_gp_order(cnt,gp_mul - no_views_eff_1/2);
			cnt = cnt+1;
			echo = echo + 1;
		if (echo < views_per_seg+1) 
			goto echoes_loop_3;

		shot = shot + 1;
	if (shot < views_per_echo*2+1) 
		goto shots_loop_3;


	goto setup_image_params;

PE_order_6_7:
	// cnt already skips the unchanged navigator train. Zero-based imaging shot s,
	// echo e, N imaging lines, L ETL, S=N/L. Values are gp_order, not physical DAC.
	// Mode 6: reverse mode-1 echo assignment separately in every imaging shot.
	// Mode 7: k(s,e) = -N/2 + s + e*S (negative to positive, interleaved).
	pe_shots = no_views_eff/views_per_seg;
	shot = 0;
new_pe_shot_loop:
	echo = 0;
new_pe_echo_loop:
	if (PE_order==6)
	{
		pe_echo_index = views_per_seg-1-echo;
		if (shot<views_per_echo)
			gp_mul = shot-views_per_echo-views_per_echo*pe_echo_index;
		else
			gp_mul = shot-views_per_echo+views_per_echo*pe_echo_index;
	}
	else
		gp_mul = -no_views_eff/2 + shot + echo*pe_shots;
	put_gp_order(cnt,gp_mul);
	cnt = cnt+1;
	echo = echo+1;
	if (echo<views_per_seg) goto new_pe_echo_loop;
	shot = shot+1;
	if (shot<pe_shots) goto new_pe_shot_loop;
	if (cnt!=no_views)
	{
		v19_error_code=74;
		goto end;
	}
	goto setup_image_params;

PE_order_2:

	// Single shot with centric order
	// Requires VIEW_ORDER_1 = CENTREOUT in reconstruction, or reordering table - cengen
	
	if (nav_on==1) {
		v19_error_code=75;
		goto end;
	}

	views_per_echo = 1;
	array_count = cnt;
	
	temp = -te_eff+1;
	
	// Single echo, sequential order
	if (views_per_seg != no_views)
	{
		v19_error_code=76;
		goto end;
	}
	
    // Centric phase encoding
	put_gp_order(0,0);
	put_gp_order(1,1);
	array_count = 2;
	do {
		ret = -1;
		if (array_count%2==1) 
			ret = 1;
		put_gp_order(array_count,ret*(array_count+1)/2);
		
		array_count = array_count + 1;
	}
	until(array_count == no_views);
	
	goto setup_image_params;
	
PE_order_5: 
	// previous combined PE_order_2 - now used with single echo only
	// Single shot with sequential order and partial Fourier option
	// Essentially te_eff is the equal to the kernel offset to be used in reconstuction!!
	// Also includes single echo when views_per_seg = 1, but te_eff should be overriden
	
	if (nav_on==1) {
		v19_error_code=75;
		goto end;
	}

	if (report_on) 
		printf("Kernel offset (for reconstruction)=%d\n", te_eff);
	views_per_echo = 1;
	array_count = cnt;
	
	temp = -te_eff+1;
	
	// Single echo, sequential order
	if (views_per_seg == 1)
		temp = -no_views/2;
	
	
    // Sequential phase encoding
	do { 
		//gp_order[array_count] = temp;
		put_gp_order(array_count, temp);
		temp = temp + 1;
		array_count = array_count + 1;
	}
	until(array_count == no_views);	
	
	goto setup_image_params;	
	

	
setup_image_params:
	// Report actual zero-PE acquisition separately; never feed it into DWI timing.
	if ((diff_on==1)&&(report_on==1))
	{
		pe_center_echo = 1;
		if (PE_order==6) pe_center_echo = views_per_seg;
		if (PE_order==7) pe_center_echo = (no_views_eff/2)/(no_views_eff/views_per_seg)+1;
		pe_center_te_us = IntToLong(te)*1000L + IntToLong(pe_center_echo-1)*IntToLong(esp)*1000L;
		printf("DWI PE=%d: first TE=%d ms, ESP=%d ms\n", PE_order, te, esp);
		printf("Zero PE at echo %d; center TE=%ld us\n", pe_center_echo, pe_center_te_us);
	}
	
	// Calculation of slice phase encoding (3D) steps
    // Sequential slice phase encoding
	temp = -no_views_2/2;
	gp_cnt = 0;
	if (pe2_centric_on==0)
	{
		do {
			//gp_var_mul2[gp_cnt] = temp;
			put_gp_var_mul2(gp_cnt, temp);
			temp = temp + 1;
			gp_cnt = gp_cnt + 1;
		}
		until(gp_cnt == no_views_2);
	}
	else if (no_views_2>2)
	// Centric slice phase encoding
	{
		//gp_var_mul2[0] = 0;
		//gp_var_mul2[1] = -1;
		put_gp_var_mul2(0, 0);
		put_gp_var_mul2(1, -1);
		gp_cnt = 2;
		do {
			ret = 1;
			if (gp_cnt%2==1) ret = -1;
			//gp_var_mul2[gp_cnt] = ret*(gp_cnt+1)/2;
			put_gp_var_mul2(gp_cnt, ret*(gp_cnt+1)/2);
			gp_cnt = gp_cnt + 1;
		}
		until(gp_cnt == no_views_2);
	}	
	else
	{
		//gp_var_mul2[0] = 0;
		//gp_var_mul2[1] = -1;
		put_gp_var_mul2(0, 0);
		put_gp_var_mul2(1, -1);		
	}	

averages_loop_setup:


/*   **************************** SET UP IMAGING PARAMETERS ***************************** */

	discard(no_discard);
	phase_increment(1);
	deg_90 = scale(90, 1000, phase_res);
	deg_360 = scale(360, 1000, phase_res);
	sample_period_l = sample_period ;
	tacq = (sample_period_l * (no_samples+no_discard))/10;
	tacq_2 = tacq/2;
	\\printf("tacq = %ld us, tacq_2 = %ld us\n",tacq, tacq_2);
	templ1 = tacq - dacmaxlong;
	if (templ1 > 0 ) {
		v19_error_code=77;
		goto end;
	}
	tacq_int = tacq;
	
/*  Define the length of the RF pulse and its associated bandwidth */

	tsel90 = rf_length[rfnum];
	tsel180 = tsel90;
	// [CRUSH-PE] Independent waveform = SEC crusher, primary RF lobe, SEC crusher.
	// Add a plateau margin for rfdelay and round RF flat time UP to an exact
	// gradient raster (tramp/10 us = five gradient clock ticks), with an even flat
	// time so the RF can be centered exactly. rfdelay is the minimum RF margin.
	crush_pre_pad = 0;
	crush_post_pad = 0;
	crush_rf_pad = 0;
	crush_rf_flat = tsel90;
	if ((crush_independent_on!=0)&&(crush_independent_on!=1))
	{
		v19_error_code=78;
		goto end;
	}
	if (crush_independent_on==1)
	{
		if ((tramp<100)||(tramp>1000)||(tramp%10!=0))
		{
			v19_error_code=79;
			goto end;
		}
		if ((crush_amp==(-32767-1))||(diff_crush_amp==(-32767-1)))
		{
			v19_error_code=80;
			goto end;
		}
		if ((tcrush<=0)||(tcrush>5000)||((IntToLong(tcrush)*50L)%IntToLong(tramp)!=0L))
		{
			v19_error_code=81;
			goto end;
		}
		if ((diff_on==1)&&((diff_tcrush<=0)||(diff_tcrush>10000)||((IntToLong(diff_tcrush)*50L)%IntToLong(tramp)!=0L)))
		{
			v19_error_code=82;
			goto end;
		}
		if (de_on==1)
		{
			v19_error_code=83;
			goto end;
		}
		temp = tramp/10;
		if ((tsel90<1)||(rfdelay<6)||(rfdelay>100))
		{ v19_error_code=84; goto end; }
		crusher_result=((IntToLong(tsel90)+2L*IntToLong(rfdelay)+IntToLong(temp)-1L)/IntToLong(temp))*IntToLong(temp);
		if (crusher_result%2L!=0L) crusher_result=crusher_result+IntToLong(temp);
		if (crusher_result>32767L)
		{ v19_error_code=85; goto end; }
		crush_rf_flat=crusher_result;
		crush_rf_pad = crush_rf_flat-tsel90;
		// [CRUSH-SCHED] Retain RF/ADC physical group-delay compensation.
		crush_pre_pad = 2*tramp+crush_rf_pad/2;
		crush_post_pad = 2*tramp+crush_rf_pad/2;
	}

	// [CRUSH-SCHED] Validate/build once, BEFORE any acquisition/validator exit.
	// Pulse k is stored at k-1. Constant diffusion-OFF retains v1.6s first
	// train baseline; DWI (including nominal b=0) uses diff_crush_amp first.
	crusher_etl = views_per_seg;
	crusher_saved_train=crush_amp;
	crusher_saved_first=diff_crush_amp;
	if ((crusher_schedule<0)||(crusher_schedule>5)||(crusher_step_pct<0)||(crusher_step_pct>1000))
	{
		v19_error_code=86;
		goto end;
	}
	if ((crusher_etl<1)||(crusher_etl>MAX_CRUSHER_ETL))
	{
		v19_error_code=87;
		goto end;
	}
	if (crusher_schedule!=0)
	{
		if ((crush_independent_on!=1)||(de_on!=0)||(echoes_to_discard!=0))
		{
			v19_error_code=88;
			goto end;
		}
		if ((no_slices<1)||(no_slices>MAX_SLICE))
		{ v19_error_code=89; goto end; }
		// The extra 1000 us is taken FROM the existing ADC window. TE, ESP
		// and the sample count do not grow. Mirror the played delay split.
		if ((no_samples<1)||(no_samples>1024)||(no_discard<0)||(no_discard>128)||(sample_period<1))
		{ v19_error_code=90; goto end; }
		crusher_adc_remaining = IntToLong(no_samples+no_discard)*IntToLong(sample_period)-3L-CRUSHER_ADC_TICKS;
		crusher_steps = crusher_adc_remaining/10000L;
		crusher_factor = crusher_adc_remaining/10L-crusher_steps*1000L;
		if (sample_period==250) crusher_factor=crusher_factor-1L;
		if ((crusher_steps<1L)||(crusher_factor<10L))
		{
			v19_error_code=91;
			goto end;
		}
	}
	if (crush_independent_on==1)
	{
		if ((crusher_max_dac<1)||(crusher_slew_dac_100us<1))
		{
			v19_error_code=92;
			goto end;
		}
		if (crusher_schedule==5)
		{
			if ((crusher_custom_count!=crusher_etl)||(crusher_custom_count<1)||(crusher_custom_count>MAX_CUSTOM_CRUSHERS))
			{
				v19_error_code=93;
				goto end;
			}
			crusher_i=0;
crusher_custom_check:
			if (crusher_custom_pct[crusher_i]==(-32767-1))
			{ v19_error_code=94; goto end; }
			crusher_i=crusher_i+1;
			if (crusher_i<MAX_CUSTOM_CRUSHERS) goto crusher_custom_check;
		}
		crusher_i=0;
crusher_schedule_loop:
		crusher_base=crusher_saved_train;
		if ((crusher_i==0)&&(diff_on==1)) crusher_base=crusher_saved_first;
		crusher_mag=IntToLong(crusher_base);
		crusher_sign=1;
		if (crusher_mag<0L) { crusher_mag=-crusher_mag; crusher_sign=-1; }
		crusher_factor=100L;
		crusher_steps=0L;
		if ((crusher_i>=1)&&((crusher_schedule==1)||(crusher_schedule==3)))
			crusher_steps=IntToLong(crusher_i)-1L;
		if ((crusher_i>=1)&&(crusher_schedule==4))
			crusher_steps=IntToLong(crusher_etl)-IntToLong(crusher_i)-1L;
		// Bound BEFORE multiplying and BEFORE adding 100.
		if (crusher_steps>0L)
		{
			if (IntToLong(crusher_step_pct)>(2147483647L-100L)/crusher_steps)
			{
				v19_error_code=95;
				goto end;
			}
			crusher_factor=100L+crusher_steps*IntToLong(crusher_step_pct);
		}
		if (crusher_schedule==5)
		{
			crusher_factor=IntToLong(crusher_custom_pct[crusher_i]);
			if (crusher_factor==-32768L) { v19_error_code=96; goto end; }
			if (crusher_factor<0L) { crusher_factor=-crusher_factor; crusher_sign=-crusher_sign; }
		}
		if (((crusher_schedule==2)||(crusher_schedule==3))&&(crusher_i%2==1))
			crusher_sign=-crusher_sign;
		// Round POSITIVE magnitude half up, then apply polarity. No clamp.
		if (crusher_mag>0L)
		{
			if (crusher_factor>(2147483647L-50L)/crusher_mag)
			{
				v19_error_code=97;
				goto end;
			}
		}
		crusher_result=(crusher_mag*crusher_factor+50L)/100L;
		if ((crusher_result>32767L)||(crusher_result>IntToLong(crusher_max_dac)))
		{
			v19_error_code=98;
			goto end;
		}
		// Both products are <= 32767*1000 after the checks above.
		if (crusher_result*100L>IntToLong(crusher_slew_dac_100us)*IntToLong(tramp))
		{
			v19_error_code=99;
			goto end;
		}
		crusher_dac[crusher_i]=crusher_result*IntToLong(crusher_sign);
		crusher_i=crusher_i+1;
		if (crusher_i<crusher_etl) goto crusher_schedule_loop;
	}
	tcrush_play = tcrush+crush_pre_pad;
	tcrush1_play = tcrush_play;
	if (diff_on==1) tcrush1_play = diff_tcrush+crush_pre_pad;
	// [V17-REVIEW] waittimer has a 16-bit argument and a 65500-tick
	// documented maximum. Validate BOTH pulse types before any acquisition.
	// Evaluate in long before multiplication or narrowing; do not wrap/clamp.
	crusher_i=0;
refocus_timer_check:
	templ1=IntToLong(tcrush_play);
	if (crusher_i==1) templ1=IntToLong(tcrush1_play);
	templ2=(templ1+IntToLong(tramp)-40L+IntToLong(rfdelay)-IntToLong(rfgate_delay-17))*10L;

	if ((report_on==1)&&(crush_independent_on==1))

where the effective phase encode time is 'tphase_effective'

This calculation will the initial gradient value in ppl units. This calculation is prone to

			if (pos_index==0)

	templ1 = (no_cest_pulses*tselmtcl + (no_cest_pulses-1)*IntToLong(inter_cest_delay)*1000L + IntToLong(mtc_tcrush+2*tramp) + IntToLong(post_cest_delay)*1000L);	
	// Calculate the inter-CEST-pulse delay 
    if(mtc_on){
		// Should remain in us!
		tr_min = tr_min + templ1;
        inter_cest_cal_min = 2*tramp + mtc_tcrush + 100;					
		// NOTE: inter_cest_delay is not a repetition time. It is the interval between the end of one pulse and the start of the next.
        inter_cest_cal = inter_cest_delay - (inter_cest_cal_min/1000L);
		if ( inter_cest_cal < 1 ) {
			v19_error_code=128;
			goto end ;
        }
    }	
#endif	
	if (tr==0) 
		tr = (tr_min*batch_slices)/1000+1;
	
	// Check if resp plateau is longer than tr_min
	// If no gate was detected t_resp_plateau would be 0
	// If gate was detected t_resp_plateau would be a positive value provided the interval is >0
	// If a positive t_resp_plateau is given in PPR then this is used instead
	if (t_resp_plateau==0) 
		t_resp_plateau = tr;
	templ1 = tr_min*IntToLong(batch_slices);
	temp = templ1/1000+1;
#ifdef MAINSGATE	
	if ((mains_gating)&&(t_resp_plateau<temp)&&(hasprinted==0))
	{
		printf("Gate interval (%dms) is shorter than min TR (%dms)\n", t_resp_plateau, temp);
		hasprinted = 1;
	}
#endif
 	if ((gating)&&(t_resp_plateau<temp)&&(hasprinted==0))
	{
		printf("Gate interval (%dms) is shorter than min TR (%dms)\n", t_resp_plateau, temp);
		hasprinted = 1;
	}
	if ((report_on == 1)&&(gating!=0)) 
		printf("Resp plateau set to %d ms\n", t_resp_plateau);
	if (report_on == 1) 
		printf("Minimum TR = %d ms\n", temp);
	tr_extend_us = ((tr*1000L/batch_slices) -tr_min); 			// [us]
	
	if ( tr_extend_us < 0 ) {
		printf("TR too short, increase to %d ms\n ", temp );
		if(batch_slices > 1) 
			v19_error_code=129;		
		goto end ;
	}
	
	if(diff_on)
	{
		tr_extend_us = ((tr*1000L/batch_slices) -tr_min); 
		\\printf("tr_extend_us = %ld us\n", tr_extend_us);
	}
	
	tr_extend_usL = tr_extend_us*10L;
	 	
	tr_min = tr_min/1000; // tr_min is the time per slice in [ms]
	
	
	// Find how many slices can fit into one resp interval assuming slice clustering is on
	temp = t_resp_plateau/tr_min;
	// The variable below will be >=1 for T2 when t_resp_plateau>=tr_min
	// and 1 for T1 when when tresp_plateau>=tr
	// clusters must be enabled!
	no_slices_per_gating_interval = temp;
	if (no_slices_per_gating_interval>batch_slices)
		no_slices_per_gating_interval = batch_slices;

	temp = t_resp_plateau/tr;
	// The variable below will be 0 for T2 when t_resp_plateau<tr
	// and >=1 for T1 when when tresp_plateau>=tr
	no_views_per_gating_interval = temp;
	
	if (no_views_per_gating_interval==0)
		no_views_per_gating_interval = no_views/views_per_seg;
	


	if ((report_on == 1)&&(gating!=0)) 
	{
		printf("Slices per gate interval = %d\n", no_slices_per_gating_interval);
		printf("Views per gate interval = %d\n",  no_views_per_gating_interval);
	 }
	
/*  Real time adjustment of slice and read compensation gradients */

	templ1 = gs_comp;
	templ1 = templ1 + (templ1*(gs_comp_scale+fse_gs_comp))/1000L; 	/*  Manual override of slice rephase lobe */
	templ2 = (templ1*IntToLong(gsp_lobe))/100L;
	templ3 = templ1 - templ2;
	if (templ2 + dacmaxlong < 0 | templ3 + dacmaxlong < 0){
		v19_error_code=130;
		goto end;
	}
	gs_rp = templ3;
	gsp_rp = (templ2*IntToLong(tref+tramp))/IntToLong(tdp+tramp);
	
	
#ifdef FSE3D
	if ((gp_sl_on>0)&&(NO_VIEWS_2>1))
	{
	/* Check if the addition of slice compensation and initial value of slice
	phase encode gradient causes an overflow */
		templ2 = gp_sl_init_var;
		templ1 = templ1 + templ2;
		if (templ1 < -32767L) {
			v19_error_code=131;
			goto end;
		}
	}	
#endif
	
	
	templ1 = gr_comp;
	templ1 = templ1 + (templ1*IntToLong(gr_comp_scale)/8)/1000L; 	/*  Manual override of switching dephase lobe */
	
	templ2 = (templ1*grp_lobe)/100L;
	templ3 = templ1 - templ2;
	if (templ2 + dacmaxlong < 0 | templ3 + dacmaxlong < 0){
		v19_error_code=132;
		goto end;
	}
	gr_dp = templ3;
	grp_dp = -(templ2*IntToLong(tdp+tramp))/IntToLong(tref+tramp);

	grp_dp_1 = 0;
	if (flow_comp_on)
	{
		templ1 = gr_comp_flow;
		templ1 = templ1 + (templ1*IntToLong(gr_comp_scale)/8)/1000L; 	/*  Manual override of switching dephase lobe */
		templ2 = (templ1*grp_lobe)/100L;
		templ3 = templ1 - templ2;
		if (templ2 + dacmaxlong < 0 | templ3 + dacmaxlong < 0){
			v19_error_code=107;
			goto end;
		}
		grp_dp_1 = -(templ2*IntToLong(tdp+tramp))/IntToLong(tref+tramp); // TODO formula was not verfified when tdp and tref made separate!
		/* also recalculate gr_flow, later to be assigned to grp_dp */
		templ3 = gr_var;
		templ4 = gr_undersample;
		templ5 = gr_oversample;
		templ3 = (templ3*templ4)/templ5;
		templ3 = templ3+templ3*IntToLong(gr_comp_scale)/8000L;
		templ3 = templ3*grp_lobe/100L;
		
		gr_flow = templ3;
		if (gr_flow + dacmaxlong < 0){
			v19_error_code=107;
			goto end;
		}
		grp_dp = -gr_flow;
	}
	
#ifdef VALIDATOR
	if (validate==1)
	{
		templ1 = IntToLong(no_views/views_per_seg) * IntToLong(no_views_2) + IntToLong(no_disacq);
		templ1 = templ1*IntToLong(no_slices/batch_slices) * IntToLong(no_averages);
		v19_error_code=133;
		goto end;
	}
#endif
	
	if (flow_comp_on) 
		templ1 = 37500L;
	else
		templ1 = 27500;
	
	// Unchanged
	
	waittimer(templ1);
	
	starttimer();
	// [V17-TIMER] Compute complete targets inside the existing setup budget.
	// Long multiplication and the echo branch must NOT follow gettimer()+250.
	post_adc_base_ticks=te_balance_bl_temp2*10L;
	post_adc_train_ticks=(te_balance_bl_temp2+te_balance_bl_esp+IntToLong(crush_post_pad-crush_pre_pad))*10L;
	G1 = grp_dp; /* if flow_comp_on = 0 */
	G2 = grp_dp_1; /* or 0 if flow_comp_on = 0 */
	
/*  Calculate the matrices for this phase-encode step */
/*  Phase-encode and refocus matrices only need be rescaled, but recalculate all for real time update of angulations */

	MR3040_SelectMatrix( ss_mat );          /*  Select a zero matrix during matrix re-calculation */
	
	CREATE_MATRIX(fse_mat,gs_on*gs_var_rescale,0,gr_on*G2)
	delay(caldelay,us);
	
	CREATE_MATRIX(fse_mat_sec,gs_on*gs_rp,0,gr_on*G1)
	delay(caldelay,us);

	CREATE_MATRIX(slice_crush,gs_on*gs_var_rescale,0,0)
	delay(caldelay,us);

	// [CRUSH-SCHED] EVERY shot/slice/row/experiment/dummy passes this setup
	// under ss_mat. Both primaries have identical slice selection. Constant
	// mode retains the v1.6 first=22, later=21 selection path.
	if (crusher_schedule==0)
	{
		CREATE_MATRIX(slice_crush_sec,gs_on*crush_independent_on*crusher_saved_train,0,0)
		delay(caldelay,us);
		crusher_play_mat=slice_crush;
		if ((crush_independent_on==1)&&(diff_on==1))
		{
			CREATE_MATRIX(first_slice_crush,gs_on*gs_var_rescale,0,0)
			delay(caldelay,us);
			CREATE_MATRIX(first_slice_crush_sec,gs_on*crusher_saved_first,0,0)
			delay(caldelay,us);
			crusher_play_mat=first_slice_crush;
		}
	}
	else
	{
		CREATE_MATRIX(slice_crush_sec,gs_on*crusher_dac[0],0,0)
		delay(caldelay,us);
		CREATE_MATRIX(first_slice_crush,gs_on*gs_var_rescale,0,0)
		delay(caldelay,us);
		crusher_base=0;
		if (crusher_etl>1) crusher_base=crusher_dac[1];
		CREATE_MATRIX(first_slice_crush_sec,gs_on*crusher_base,0,0)
		delay(caldelay,us);
		crusher_play_mat=slice_crush;
	}
	
	CREATE_MATRIX(diff_mat, -diff_slice, -diff_phase, -diff_read)
	delay( caldelay, us);
	
#ifdef MULTIPRESAT
	CREATE_MATRIX(presat_mat_sec, sat_on*sat_gr_amp, sat_on*sat_gr_amp, sat_on*sat_gr_amp)
	delay(caldelay,us);
#endif

#ifdef CHESS
	CREATE_MATRIX( chess_mat, -gs_chess_amp, 0 ,0)
	delay( caldelay, us);	
#endif

#ifdef MTC
	CREATE_MATRIX( mtc_mat, -mtc_on*mtc_gr_amp, -mtc_on*mtc_gr_amp, -mtc_on*mtc_gr_amp)
	delay( caldelay, us);	
#endif


/* For narrow band systems the observe modulator level can be 0 or 4 ie on or off
full control is provided in the instrument panel 0db to - 40db . So if the RF is on
set the observe modulator level to 4 */

	rf_on = 0 ;
	if ( obs_mod_level > 0 ) 
		rf_on = 4 ;

	// The minimum time returned by hostrequest() is 1 [ms] even if nothing to update
	time = hostrequest(); // 91 ticks
	if (time==1)
		time = 0;

	// [CRUSH-SCHED] DWI setup creates the same six FSE/crusher matrices as
	// v1.6. Diffusion-OFF scheduled mode needs two more: reject a setup
	// overrun here, BEFORE the excitation/ADC, rather than lengthening TR.
	// Reserve 550 us for timer/check overhead within the 3000 us setup window.
	// MAN 4.8.1 documents timing-function overhead of 360.2 +/- 5.0 us;
	// the assignment/comparisons also consume this allowance. Target timing
	// still requires vendor verification. Reject signed timer overrange.
	crusher_setup_ticks = gettimer();
	if ((crusher_setup_ticks<0)||(crusher_setup_ticks>24500))
	{
		v19_error_code=134;
		goto end;
	}
	
	
	if (v19_mode==1) goto v19_mat_window;
	waittimer(30000); // 3000 us; included in the 11350-us TR overhead

	resync();

/*   ************************************ SEQUENCE STARTS HERE ********************************************** */

	// T2 gating - do specific subset of slices ETs in each resp interval
	// T1 gating - do batch_slices echo trains (ETs) in each resp interval
	// T1 with long resp interval - fit multiple TRs in each resp interval (do a specified number of VPS in each r.i.)
	gating_condition_1 = ((current_slice-1)%no_slices_per_gating_interval==0); // T2, 82 ticks
	// Check if both conditions are true (i.e., slice is at the start of the batch and the view is good for triggering)

#ifdef MAINSGATE	
	if (mains_gating==1)
	{
		gating_condition = (current_slice==1); // T2, 82 ticks
		
		if (gating_condition)
		{		
			// wait for HI
			do{
				noop;
				hostrequest();
			} until input#1!=0;
			
			// If HI detected, wait for LO
			if (input#1==1)
				do{
					noop;
					hostrequest();
				} until input#1!=1;
			
			if (post_mains_gate_delay>0)
				delay(post_mains_gate_delay, ms);
			// UNTESTED
		}
	}	
	// Insert secondary mains triggering delays
	if (current_slice==3)
		delay(93, ms);
	if (current_slice==6)
		delay(140, ms);
	if (current_slice==8)
		delay(93, ms);
	if (current_slice==11)
		delay(140, ms);
	if (current_slice==13)
		delay(93, ms);
	if (current_slice==16)
		delay(140, ms);
	if (current_slice==18)
		delay(93, ms);
	if (current_slice==21)
		delay(140, ms);
	if (current_slice==23)
		delay(93, ms);
	if (current_slice==26)
		delay(140, ms);
#endif

	if (gating)
	{
		if (gating_mode==1) 
			gating_condition = gating_condition_1; // T2
		else if (gating_mode==2) 
			gating_condition = gating_condition_1; // T1 and T2 are now identical
		else 
			gating_condition = -1; // trigger each echo train (=slice) - e.g. for cardiac
		//output(19);  // Select Userport
		if (gating_condition)
		{
			do{
				noop;
				hostrequest();
			} until input#0!=0;
		}
	}

	
/*  Selective 90 degree pulse in presence of slice select gradient */

	echo_cnt = 0;
	total_echo_cnt = 0; // counts all echo, including first discarded ones
	resync();
	phase(phase_90);                        /*  90x pulse */	
	MR3040_Clock(clock);
	
#ifdef MTC
    if(mtc_on){
		cest_count = 0;								\\ CEST pulse loop counter
		cest_loop:
			MR3040_SetList( mtc_list, ALL_CHANNELS );
			MR3040_SelectMatrix( mtc_mat );
			frequency_buffer(5);
			reset_frequency();
#ifdef PDD	
			if (use_pdd>0)
				userout(pdd_tx_mask);
#endif
			if (mtc_gaussian)
			{
				temp_mac = 100 - (22+warmup);
				delay(temp_mac,us);
				Set_Board_Multipliers( mtc_mul,0, rbd[mtc_rfnum]);
				Set3031Address(rad[mtc_rfnum],rbd[mtc_rfnum]+4);
				Set3031Waits(rwt[mtc_rfnum],rbd[mtc_rfnum]+4);
				Append_Boards_Ready( rbd[mtc_rfnum]+4 );
				rfampon(0);
				delay(warmup,us);
				rfon(rf_on);
				MR3031_go();
				delay(tselmtc_ms, ms);
				rfon(0);
			}
			else
			{
				MR3031_setup(pf1,"rf_analog_on",0,0,0,mtc_mul);
				rfampon(0);
				delay(warmup,us);
				rfon(rf_on);
				MR3031_go();
				delay(tselmtc_ms, ms);				
				rfoff();
				MR3031_setup(pf1,"rf_off",0,0,0,0);
				MR3031_go();
			}
			delay(100,us);
#ifdef PDD	
			if (use_pdd>0)
				userout(pdd_rx_mask);
#endif
			MR3040_Clock(clock);                    \\ Extra 3040 call for 3040 syncronisation
			delay(tramp,us);
			delay(mtc_tcrush,us);
			starttimer();
				ret = tramp*10;
				frequency_buffer(0);
				reset_frequency();
				cest_count = cest_count + 1;
			waittimer(ret);
			if (cest_count < no_cest_pulses) {               \\ CEST pulse loop
				delay(inter_cest_cal, ms);
				goto cest_loop;
			}
			if (post_cest_delay>0)
				delay(post_cest_delay, ms);
        }
#endif	
	
	
#ifdef MULTIPRESAT
 	if ((sat_on)&&(no_PB>0))
	{
		PB_index = 0;
		do {
			if (PB_on[PB_index]==1) 
			{
				starttimer(); 
					sat_channel = PB_channel[PB_index];
					gs_sat = PB_gs_sat[PB_index];
 					sat_on_ar[0] = -(sat_channel == CHANNEL_S); 
					sat_on_ar[1] = -(sat_channel == CHANNEL_P); 
					sat_on_ar[2] = -(sat_channel == CHANNEL_R); 
					CREATE_MATRIX(presat_mat, sat_on_ar[0]*gs_sat, sat_on_ar[1]*gs_sat, sat_on_ar[2]*gs_sat)					
					delay(caldelay,us);  
					MR3040_SetList( sat_list, ALL_CHANNELS);
					MR3040_SelectMatrix( presat_mat );
					MR3040_Clock(clock);  
					frequency_buffer(3);     \\ need to overwrite the buffer here
					frequency(MHz, kHz+PB_kHz[PB_index], Hz+PB_Hz[PB_index], rx1MHz);
					offset_frequency(PB_freq[PB_index]);
					reset_frequency();
				waittimer(5200); // 4936
				MR3040_Start(ALL_CHANNELS);
				
#ifdef PDD
				if (use_pdd>0)
					userout(pdd_tx_mask);
#endif			
				NEWSHAPE_SETUP(rfnum_sat, p_sat_mul)
				
				rfampon(0);
				if (warmup>10) delay(warmup,us);
				rfon(rf_on);
				MR3031_go();
				
				starttimer();
					// Presat to apply on all OTHER axes only
					
					sat_on_ar[0] = -!sat_on_ar[0];
					sat_on_ar[1] = -!sat_on_ar[1];
					sat_on_ar[2] = -!sat_on_ar[2];
					CREATE_MATRIX(presat_mat_sec, sat_on_ar[0]*sat_gr_amp, sat_on_ar[1]*sat_gr_amp, sat_on_ar[2]*sat_gr_amp)
					ret = tsel_sat-500;
				waittimer(5000);
				delay(ret,us);
				
				rfon(0);
				MR3040_CONTINUE(ALL_CHANNELS);
#ifdef PDD
				if (use_pdd>0)
					userout(pdd_rx_mask);
#endif	
				delay(tramp,us);
				delay(sat_tcrush,us);
				MR3040_CONTINUE(ALL_CHANNELS);
				
				starttimer();
					ret = tramp*10;
					frequency_buffer(0);    
					reset_frequency();
				waittimer(ret);
				delay(inter_PB_delay,us);
			}
			PB_index = PB_index+1;						
		} until (PB_index==MAX_PRESAT_BLOCKS);
    }
	
	
	
#endif	

#ifdef CHESS
	if (chess_on)
	{
		// Duration = 30+t_chess+40+2*tramp+tcrush_chess [us]
		starttimer();
			frequency_buffer(4);
			reset_frequency();
		waittimer(300);
#ifdef PDD
		if (use_pdd>0)
			userout(pdd_tx_mask);
#endif		
		if (chess_gauss>0)
		{
			NEWSHAPE_SETUP(chess_rfnum, chess_mul)
			
			rfampon(0);
			if (warmup>10) delay(warmup,us);
			rfon(rf_on);
			MR3031_go();
			
			starttimer();
				// Calculate crusher parameters
				MR3040_SetList( chess_list, CHANNEL_S );
				MR3040_SelectMatrix( chess_mat );
				MR3040_Clock(clock); 
				ret = t_chess_gauss-1; // [ms]
			waittimer(10000);
			delay(ret, ms);
			
			rfon(0);
		}
		else
		{
			MR3031_setup(pf1,"rf_analog_on",0,0,0,chess_mul);
			rfampon(0);
			if (warmup>10) delay(warmup,us);
			rfon(rf_on);
			MR3031_go();
			starttimer();
				// Calculate crusher parameters
				MR3040_SetList( chess_list, CHANNEL_S );
				MR3040_SelectMatrix( chess_mat );
				MR3040_Clock(clock); 
				ret = t_chess*10;
			waittimer(ret);
			rfoff();
			MR3031_setup(pf1,"rf_off",0,0,0,0);
			MR3031_go();
		}		
#ifdef PDD		
		if (use_pdd>0)
			userout(pdd_rx_mask);
#endif		
		MR3040_Start(CHANNEL_S);
		// Restore slice frequency within while ramping the gradient
		starttimer();
			ret = tramp*10;
			frequency_buffer(0);    
			reset_frequency();
		waittimer(ret);		
		delay(tcrush_chess,us);
		delay(tramp,us);
	}	
#endif
	
	MR3040_SetList( slice_list, CHANNEL_S); 	// 33 ticks
	MR3040_SetList( read_pre_list, CHANNEL_R ); // 33 ticks
	MR3040_SelectMatrix( fse_mat ); 			// 20 ticks
	MR3040_Clock(clock); // makes images worse. Or better? 22 ticks
	MR3040_Start( CHANNEL_S|CHANNEL_R );
	
	// 90 degree pulse

	// there is a default ~17us delay between the beginning of the gate and start of the pulse
	// rfdelay is the delay from end of ramp to start of rf pulse, not the gate
	// TODO this is not timed
 	temp_mac = tramp-40+rfdelay-(rfgate_delay-17); // -40 gives 0 delay; takes 40 ticks

	starttimer();
#ifdef PDD
		if (use_pdd>0)
			userout(pdd_tx_mask);
#endif	
		ret = temp_mac*10;
	waittimer(ret);
	NEWSHAPE_SETUP(rfnum, p90_mul)
	rfampon(0); // this opens the gate
 	if (warmup>10) 
		delay(warmup,us);
	starttimer();
		ret = (rfgate_delay-17)*10;
	waittimer(ret);
	rfon(rf_on);
	MR3031_go();
	ret = tsel90 - 3-rfdelay;
	delay(ret, us);
	MR3040_CONTINUE(CHANNEL_S|CHANNEL_R);	
	starttimer();
		ret = rfdelay*10; // Note that the minimum waittimer delay is 50 clocks
	waittimer(ret);
		rfon(0); 
#ifdef PDD
		if (use_pdd>0)
			userout(pdd_rx_mask);
#endif
		// Specsim-s interpretation of waittimer is wrong with multiple waittimers, it resets the timer after waittimer is called
		templ3 = IntToLong(tramp*3+tref);
		templ3 = templ3*10L;

		resync();
		phase(phase_180-phcor0);      \\ 180 y pulse
		if (diff_on == 0)
		{
			post_90_delay0L = te_balance_al + IntToLong(-21-6+post_90_delay1); // 6us is an empirically detected delay
			post_90_delay0L = post_90_delay0L*10L;
			pre_90_delay0L = post_90_delay0L;
			
		}
		else
		{
			// [TWO-TE FIX #3] subtract the first-180 crusher surplus (tcrush1 - tcrush)
			// now physically played by slice_180_refocus_diff and the this_tcrush delays
			post_90_delay0L = te_balance_al + IntToLong(-21-6+post_90_delay1) - sm_delta_us - diff_tramp - IntToLong(tcrush1 - tcrush);
			post_90_delay0L = post_90_delay0L*10L - extra_delta_us - 185;
			pre_90_delay0L = post_90_delay0L;
			\\printf("Post90 delay = %ld\n",post_90_delay);
		}
		
		if (flow_comp_on) 
			post_90_delay0L = post_90_delay0L+IntToLong((20*tramp)+(5*tsel90));
		if(templ3>49000L)
		{
			ret = templ3/10-4800;
			// 25 is the time taken by if statement before delay32 below

  VERSION 1.0
  TWO-ECHO-TIME MODIFICATION - CHANGE LOG AND VALIDATION NOTES
  Derived from FSE_dwi_CPMG_non_CPMG.ppl (FSENFP.PPL, v2.7.7c/DWI, Garipov/AJVB).
  Purpose: allow a long diffusion-weighted first echo (TE = te) followed by a SHORT
  imaging echo train, by permitting diff_tcrush (first-180 time budget) > tcrush
  (imaging-180 crushers). This restores the sequence's intended capability, which was
  blocked by a temporary guard.

  ------------------------------------------------------------------------------
  WHAT WAS CHANGED (search the file for the tags):

  [TWO-TE MOD #1]  (~ line 1786, "directions_loop" block)
      Removed the two hard guards that aborted the scan:
        - if (diff_tcrush != tcrush) ... goto end;   (forced the two crushers equal)
        - if (diff_tcrush >  tcrush)  ... goto end;   (blocked the desired direction)
      Replaced with a non-aborting sanity warning if diff_tcrush < tcrush. The original
      lines are preserved in a comment. This is the ONLY change that alters behaviour.

  [TWO-TE MOD #2]  (~ line 1820, validator worst-case block)
      No code change. Added comments documenting WHY the existing worst-case forcing
      (tcrush1 = diff_tcrush during validate==1) makes diff_tcrush > tcrush safe to
      validate, and replaced the author's "TODO this is still wrong" note with an
      explicit instruction to verify on the console.

  ------------------------------------------------------------------------------
  WHY THIS IS THE CORRECT (AND MINIMAL) CHANGE - read before editing further:

  In this sequence the slice-refocus crusher gradient list (slice_180_refocus, built at
  ~line 1451 with POSPULSE(tsel90 + tcrush + tcrush)) uses tcrush for EVERY 180, first
  and imaging alike. The heavy dephasing on the FIRST 180 is provided by the diffusion
  gradients G1/G2 that straddle it, NOT by a separate large slice-crusher lobe.
  diff_tcrush therefore acts as a TIME BUDGET reserved around the first 180 (it enters
  min_pre, min_post and big_delta_min via tcrush1), while the imaging echo spacing is set
  by tcrush (via te_a / te_b and the slice_180_refocus list). Decoupling the two echo
  times is thus achieved simply by allowing diff_tcrush > tcrush - which is what MOD #1
  does. The PARAMLIST defaults (tcrush = 10 us, diff_tcrush = 2000 us) confirm this was
  the original design intent.

  ------------------------------------------------------------------------------
  RECOMMENDED PPR SETTINGS FOR A TWO-TE DW-FSE RUN:
      diff_on       = 1            ; enable diffusion
      PE_order      = 1            ; multi-echo DWI order (NOT 5, which forces ETL = 1)
      views_per_seg = >1           ; the imaging echo train length (e.g. 8 or 16)
      te            = <long>       ; the diffusion-weighted first-echo time (effective TE)
      tcrush        = <small>      ; imaging-180 crusher -> sets the SHORT train spacing
      diff_tcrush   = <large>      ; first-180 time budget (>= tcrush; e.g. 2000)
      sm_delta, big_delta, b-values, no_diff_dir, directions : as required
  Note: the uploaded PPR had diff_on = 0, tcrush = 400, diff_tcrush = 50 (reversed).
  For two-TE DWI set diff_tcrush >= tcrush as above.

  [ACQ-TABLE] Explicit diffusion acquisition table / 2026 (tagged [ACQ-TABLE])
  ============================================================================
  Purpose: replace the rectangular (no_b_steps x no_diff_dir) diffusion grid with
  an explicit table with ONE ROW PER ACQUIRED DIFFUSION VOLUME, so that b=0 need
  not be repeated once per direction, each shell may carry a different number of
  directions, and extra b=0 volumes may be inserted anywhere in the order.

  Interface (PARAMLIST / PPR):
      no_diff_acq            number of table rows (1 .. MAX_DIFF_ACQ = 512)
      acq_b[512]             requested b, s/mm^2   (used when b_input_mode = 1)
      acq_grad[512]          gradient DAC          (used when b_input_mode = 0;
                             written by the setup b->DAC search when mode = 1)
      acq_x/acq_y/acq_z[512] direction components, 1000-normalised
  Superseded: no_b_steps, b_steps_array[16], b_grads_array[16], no_diff_dir,
  diff_grad_dir[48], and the runtime counters b_step_cnt / diff_dir_cnt (now the
  single index diff_acq_cnt). MAX_BVALS / MAX_DIRECTIONS / MAX_DIRECTIONSX3 are
  replaced by MAX_DIFF_ACQ.

  Mapping from an old PPR: an old configuration with no_b_steps = B and
  no_diff_dir = D becomes no_diff_acq = B*D rows, row (i*D + j) carrying
  b_steps_array[i] / b_grads_array[i] and direction j - i.e. exactly the order
  the old nested loop acquired them in, so the MRD experiment index is unchanged.

  Deliberately PRESERVED:
    - the b->DAC binary search, b_kfac kernel, max_diff_grad = 30000 cap,
      diff_grad_scale back-conversion and achieved-b reporting (F3), all still
      executed once per row in the setup section, never during acquisition;
    - the DAC = 1 workaround for b = 0 in b-value mode. A b = 0 row must
      therefore carry a UNIT direction; the normalisation check now runs over
      every table row, so a literal X = Y = Z = 0 row is rejected in b-value mode
      instead of silently turning the DAC = 1 lobe into a true zero gradient.
      DAC mode (b_input_mode = 0) is unchanged and still takes acq_grad
      literally, including 0;
    - diff_on timing (FIX #2): tcrush1 = diff_tcrush, min_pre/min_post,
      big_delta, extra_delta and the first-180 budget follow diff_on, not the
      gradient amplitude, so the surviving b = 0 volume keeps the timing it had;
    - the outer experiment repetition: the test is no_experiments % no_diff_acq,
      NOT equality. The acquisition loop only re-enters experiment_loop after a
      complete pass over the table, so the acquired volume count is always a
      multiple of no_diff_acq; the modulo test makes that multiple equal
      no_experiments, exactly as the old no_b_steps*no_diff_dir test did. This
      keeps TR_array, Dixon W-F (needs no_experiments >= 3), MTC array indexing

    - the runtime scale(acq_grad[i], diff_grad_scale, 100), so the interactively
